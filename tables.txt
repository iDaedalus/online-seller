TMDB

create table tmdb_details_cache (
  id bigint generated always as identity primary key,
  tmdb_id integer not null,
  media_type text not null,
  language text not null,
  data jsonb not null,
  refreshed_at timestamptz not null default now(),
  expires_at timestamptz not null default (now() + interval '7 days'),
  constraint tmdb_details_cache_unique unique (tmdb_id, media_type, language)
);

create table tmdb_season_cache (
  id bigint generated always as identity primary key,
  tmdb_id integer not null,
  season integer not null,
  language text not null,
  data jsonb not null,
  refreshed_at timestamptz not null default now(),
  expires_at timestamptz not null default (now() + interval '7 days'),
  constraint tmdb_season_cache_unique unique (tmdb_id, season, language)
);

BERKAS
create table berkas_cache (
  id uuid primary key default gen_random_uuid(),
  tmdb_id text not null,
  media_type text not null,
  season text not null default '',
  episode text not null default '',
  stream_urls jsonb not null,
  subtitles jsonb not null default '[]',
  refreshed_at timestamptz not null default now(),
  expires_at timestamptz not null default (now() + interval '3 hours'),
  unique (tmdb_id, media_type, season, episode)
);

create index idx_berkas_cache_lookup
  on berkas_cache (tmdb_id, media_type, season, episode, expires_at);

HOLLY
create table holly_movie_cache (
    id          bigserial primary key,
    tmdb_id     integer not null,
    media_type  text not null,
    season      text not null default '',
    episode     text not null default '',
    embeds      jsonb not null default '[]'::jsonb,
    sources     jsonb not null default '[]'::jsonb,
    created_at  timestamptz default now(),
    constraint holly_movie_cache_unique
        unique (tmdb_id, media_type, season, episode)
);

create index idx_holly_movie_cache_created_at
on holly_movie_cache (created_at desc);

MOVIEBOX APP/WEB
create table moviebox_cache (
  id uuid primary key default gen_random_uuid(),
  tmdb_id text not null,
  title text not null,
  release_date text not null,
  media_type text not null,
  dubs jsonb not null default '[]'::jsonb,
  manual boolean not null default false,
  created_at timestamptz default now(),
  unique(tmdb_id, media_type)
);
create index idx_moviebox_cache_tmdb on moviebox_cache(tmdb_id, media_type);


create table moviebox_downloads_cache (
  id uuid primary key default gen_random_uuid(),
  play_count integer not null default 0,
  error_count integer not null default 0,
  tmdb_id text not null,
  media_type text not null,
  season text not null default '',
  episode text not null default '',
  dub text not null default 'orig',
  type integer not null default 0,
  downloads jsonb not null default '[]'::jsonb,
  created_at timestamptz default now(),
  refreshed_at timestamptz default now(),
  expires_at timestamptz default (now() + interval '45 hours'),
  unique(tmdb_id, media_type, season, episode, dub, type)
);
