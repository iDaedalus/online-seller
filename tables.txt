// CORS preflight
app.options("/proxy", async (request, reply) => {
  const origin = request.headers.origin || "";

  if (!ALLOWED_ORIGINS.includes(origin)) {
    return reply.code(403).send();
  }

  reply
    .header("Access-Control-Allow-Origin", origin)
    .header("Access-Control-Allow-Methods", "GET,HEAD,OPTIONS")
    .header(
      "Access-Control-Allow-Headers",
      "Range, Authorization, If-Range, If-None-Match, If-Modified-Since, Content-Type"
    )
    .header("Access-Control-Expose-Headers", "*")
    .header("Access-Control-Max-Age", "86400")
    .header("Vary", "Origin");

  return reply.code(204).send();
});

app.route({
  method: ["GET", "HEAD"],
  url: "/proxy",

  handler: async (request, reply) => {
    const { url } = request.query;

    if (!url) {
      return reply.code(400).send("Missing url");
    }

    const origin = request.headers.origin || "";
    const referer = request.headers.referer || "";

    const allowed =
      ALLOWED_ORIGINS.includes(origin) ||
      ALLOWED_ORIGINS.some((o) => referer.startsWith(o + "/"));

    if (!allowed) {
      return reply.code(403).send({
        success: false,
        error: "Forbidden",
      });
    }

    try {
      const headers = { ...BASE_HEADERS };

      for (const name of FORWARD_HEADERS) {
        const value = request.headers[name];
        if (value) {
          headers[name] = value;
        }
      }

      const upstream = await fetch(url, {
        method: request.method,
        headers,
        redirect: "follow",
      });

      reply.code(upstream.status);

      for (const name of RESPONSE_HEADERS) {
        const value = upstream.headers.get(name);
        if (value) {
          reply.header(name, value);
        }
      }

      if (origin) {
        reply.header("Access-Control-Allow-Origin", origin);
        reply.header("Vary", "Origin");
      }

      reply.header("Access-Control-Expose-Headers", "*");

      return reply.send(upstream.body);
    } catch (err) {
      console.error(err);

      return reply.code(500).send({
        success: false,
        error: err.message,
      });
    }
  },
});

app.listen({
  host: "127.0.0.1",
  port: 3000,

function decryptData(data) {
  const input = Buffer.from(data, "base64url");

  const iv = input.subarray(0, 12);
  const ciphertext = input.subarray(12, input.length - 16);
  const authTag = input.subarray(input.length - 16);

  const decipher = crypto.createDecipheriv("aes-256-gcm", AES_KEY, iv);
  decipher.setAuthTag(authTag);

  return Buffer.concat([
    decipher.update(ciphertext),
    decipher.final(),
  ]).toString("utf8");
}
